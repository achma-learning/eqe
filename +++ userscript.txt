    function registerInterval(intervalId) {
        cleanupRegistry.intervals.add(intervalId);
        return intervalId;
    }

    function registerTimeout(timeoutId) {
        cleanupRegistry.timeouts.add(timeoutId);
        return timeoutId;

    function registerObserver(observer) {
        cleanupRegistry.observers.add(observer);
        return observer;
    }

    function registerEventListener(element, event, handler, options) {
        cleanupRegistry.eventListeners.push({ element, event, handler, options });
        element.addEventListener(event, handler, options);
    }

    function cleanup() {
        console.log('[Auto-Advance] Cleaning up resources...');

        cleanupRegistry.intervals.forEach(id => clearInterval(id));
        cleanupRegistry.intervals.clear();

        cleanupRegistry.timeouts.forEach(id => clearTimeout(id));
        cleanupRegistry.timeouts.clear();

        cleanupRegistry.observers.forEach(obs => obs.disconnect());
        cleanupRegistry.observers.clear();

        cleanupRegistry.eventListeners.forEach(({ element, event, handler, options }) => {
            element?.removeEventListener(event, handler, options);
        });
        cleanupRegistry.eventListeners = [];

        clearTimer();
        stopCountdown();

        if (state.questionLoadDebounce) {
            clearTimeout(state.questionLoadDebounce);
            state.questionLoadDebounce = null;
        }

        console.log('[Auto-Advance] Cleanup complete');
    }

    function playNotificationSound() {
        try {
            const audioContext = new (window.AudioContext || window.webkitAudioContext)();
            const oscillator = audioContext.createOscillator();
            const gainNode = audioContext.createGain();

            oscillator.connect(gainNode);
            gainNode.connect(audioContext.destination);

            oscillator.type = 'sine';
            oscillator.frequency.setValueAtTime(440, audioContext.currentTime);
            gainNode.gain.setValueAtTime(0, audioContext.currentTime);
            gainNode.gain.linearRampToValueAtTime(0.3, audioContext.currentTime + 0.01);
            oscillator.start(audioContext.currentTime);
            gainNode.gain.exponentialRampToValueAtTime(0.001, audioContext.currentTime + 0.5);
            oscillator.stop(audioContext.currentTime + 0.5);
        } catch (e) {
            console.warn('[Auto-Advance] Could not play notification sound:', e);
        }
    }

    function scanAndSaveCourses() {
        const courseLinks = document.querySelectorAll('a[href*="/dashboard/course/"]');
        const uniqueCourses = [];
        const seenNames = new Set();
        const seenIds = new Set();

        courseLinks.forEach(link => {
            const id = link.href.split('/').pop();
            const nameElement = link.querySelector('h4') || link.querySelector('h3') || link.querySelector('span');
            let name = nameElement ? nameElement.innerText.trim() : "Unknown Module";
            name = name.replace(/^(key=\d+|press=\[\d+\])\s+/, '');
            const normalizedName = name.toLowerCase();

            if (!seenIds.has(id) && !seenNames.has(normalizedName)) {
                seenIds.add(id);
                seenNames.add(normalizedName);
                uniqueCourses.push({ name, url: link.href, id });
            }
        });

        if (uniqueCourses.length > 0) GM_setValue('saved_courses', uniqueCourses);
    }

    function decorateDashboardModules() {
        if (!window.location.href.includes('/dashboard')) return;
        const courses = GM_getValue('saved_courses', []);
        if (courses.length === 0) return;

        document.querySelectorAll('a[href*="/dashboard/course/"]').forEach(link => {
            const id = link.href.split('/').pop();
            const courseIndex = courses.findIndex(c => c.id === id);
            if (courseIndex !== -1) {
                const num = courseIndex + 1;
                const nameElement = link.querySelector('h4') || link.querySelector('h3') || link.querySelector('span');
                if (nameElement) {
                    const desiredHtml = `<span style="color: #1793d1;">press=[${num}]</span> ${courses[courseIndex].name}`;
                    if (nameElement.innerHTML !== desiredHtml) nameElement.innerHTML = desiredHtml;
                }
            }
        });
    }

        });

        Object.assign(panel.style, {
            backgroundColor: bg, color: text,
            padding: '24px', borderRadius: '16px',

                        </tr>
                    </thead>
                    <tbody>
        `;

        courses.forEach((c, i) => {
            const isCurrent = window.location.href.includes(c.id);
            tableHtml += `
                <tr class="eqe-course-row" data-url="${c.url}" style="cursor: pointer; border-bottom: 1px solid ${border}; ${isCurrent ? 'background: rgba(59, 130, 246, 0.1);' : ''}">
                    <td style="padding: 12px 8px; font-weight: bold;">${i + 1}</td>
                    <td style="padding: 12px 8px;">${c.name}</td>
                </tr>
            `;
        });

        tableHtml += `</tbody></table></div>
            <p style="margin: 20px 0 0 0; font-size: 12px; opacity: 0.7; text-align: center;">Press 1-9 to select • [Esc] to close</p>`;

        overlay.appendChild(panel);
        document.body.appendChild(overlay);

        const closeOnEsc = (e) => {
            if (e.key === 'Escape') {
                overlay.remove();
                window.removeEventListener('keydown', closeOnEsc);
                window.removeEventListener('keydown', handleNumberNav);
            }
        };

        const handleNumberNav = (e) => {
            if (/^[1-9]$/.test(e.key) && !e.shiftKey && !e.ctrlKey && !e.altKey) {
                const num = parseInt(e.key);
                if (num > 0 && num <= courses.length) {
                    window.location.href = courses[num - 1].url;
                    overlay.remove();
                    window.removeEventListener('keydown', closeOnEsc);
                    window.removeEventListener('keydown', handleNumberNav);
                    e.preventDefault();
                    e.stopImmediatePropagation();
                }
            }
        };

        window.addEventListener('keydown', closeOnEsc);
        window.addEventListener('keydown', handleNumberNav);

        panel.querySelectorAll('.eqe-course-row').forEach(row => {
            row.onclick = () => { window.location.href = row.dataset.url; overlay.remove(); };
            row.onmouseover = () => row.style.backgroundColor = 'rgba(59, 130, 246, 0.05)';
            row.onmouseout = () => {
                const isCurrent = window.location.href.includes(row.dataset.url.split('/').pop());
                row.style.backgroundColor = isCurrent ? 'rgba(59, 130, 246, 0.1)' : 'transparent';
            };
        });

        overlay.onclick = (e) => {
            if (e.target === overlay) {
                overlay.remove();
                window.removeEventListener('keydown', closeOnEsc);
                window.removeEventListener('keydown', handleNumberNav);
            }
        };
    }

    }

    function togglePomotroid() {
        if (pomoState.running) {
            // Pause
            pomoState.running = false;
            clearInterval(pomoState.interval);
            pomoState.interval = null;
            updatePomotroidVisuals();
        } else {
            // Start / resume
            if (pomoState.remaining <= 0) {
                // Reset before starting
                pomoState.remaining = POMO_TOTAL_SECONDS;
            }
            pomoState.running = true;
            pomoState.interval = setInterval(() => {
                pomoState.remaining--;
                updatePomotroidVisuals();

                if (pomoState.remaining <= 0) {
                    clearInterval(pomoState.interval);
                    pomoState.interval = null;
                    pomoState.running = false;
                    onPomotroidFinished();
                }
            }, 1000);
            updatePomotroidVisuals();
        }
    }

    function updatePomotroidVisuals() {
        const ring = document.getElementById('eqe-pomo-ring');
        const text = document.getElementById('eqe-pomo-text');
        if (!ring || !text) return;

        const pct = Math.max(0, (pomoState.remaining / POMO_TOTAL_SECONDS) * 100);
        ring.style.strokeDashoffset = (100 - pct).toFixed(2);

        // Text: ceil so 39m55s still shows 40 until it crosses the minute boundary
        const minsLeft = Math.ceil(pomoState.remaining / 60);
        text.textContent = minsLeft > 0 ? minsLeft : '0';

        // Colour shifts: green > 50%, amber 20-50%, red <20%
        if (!pomoState.running) {
            ring.style.stroke = 'rgba(255,255,255,0.45)'; // paused — greyed
        } else if (pct > 50) {
            ring.style.stroke = '#ff4e4d'; // Pomotroid coral
        } else if (pct > 20) {
            ring.style.stroke = '#f59e0b'; // amber warning
        } else {
            ring.style.stroke = '#ef4444'; // urgent red
        }

        // Update button tooltip
        const btn = document.getElementById('eqe-btn-pomo');
        if (btn) {
            const m = Math.floor(pomoState.remaining / 60);
            const s = (pomoState.remaining % 60).toString().padStart(2, '0');
            btn.title = pomoState.running
                ? `Session Timer — ${m}:${s} left (click to pause)`
                : `Session Timer — ${m}:${s} left (click to resume)`;
        }
    }

    function onPomotroidFinished() {
        const ring = document.getElementById('eqe-pomo-ring');
        const text = document.getElementById('eqe-pomo-text');
        if (ring) ring.style.stroke = '#10b981'; // brief green flash on finish
        if (text) text.textContent = '✓';

        playNotificationSound();
        showToast('🍅 Session complete! Take a break.', 'success');

        // Reset visuals after 3 s
        setTimeout(() => {
            pomoState.remaining = POMO_TOTAL_SECONDS;
            updatePomotroidVisuals();
            if (text) text.textContent = '40';
        }, 3000);
    }

    function updateDynamicIslandTimer() {
        const island = document.getElementById('eqe-dynamic-island');
        if (!island) return;

        const questionGradient = island.dataset.questionGradient;
        const answerGradient = island.dataset.answerGradient;
        const lowTimeGradient = island.dataset.lowTimeGradient;

        if (!config.autoAdvanceEnabled || !state.currentPhase) {
            island.style.opacity = '0';
            island.style.transform = config.islandLeft === '50%' ? 'translateX(-50%) scale(0.8)' : 'scale(0.8)';
            const hideTimeoutId = registerTimeout(setTimeout(() => {
                island.style.display = 'none';
                cleanupRegistry.timeouts.delete(hideTimeoutId);
            }, 400));
            return;
        }

        if (island.style.display === 'none') {
            island.style.display = 'flex';
            const showTimeoutId = registerTimeout(setTimeout(() => {
                island.style.opacity = '1';
                island.style.transform = config.islandLeft === '50%' ? 'translateX(-50%) scale(1)' : 'scale(1)';
                cleanupRegistry.timeouts.delete(showTimeoutId);
            }, 10));
        }

        const phase = state.currentPhase === 'question' ? 'Question' : 'Answer';
        const emoji = state.currentPhase === 'question' ? '❓' : '✅';
        const isLowTime = state.timeRemaining <= 3 && state.timeRemaining > 0;

        let currentGradient = state.currentPhase === 'question' ? questionGradient : answerGradient;
        if (isLowTime) currentGradient = lowTimeGradient;

        island.style.background = currentGradient;

        if (state.timeRemaining > 0) {
            island.innerHTML = `
                <span style="margin-right: 8px; font-size: 18px;">${emoji}</span>
                <span style="opacity: 0.9; margin-right: 8px;">${phase}</span>
                <span style="font-size: 20px; font-weight: 700; font-variant-numeric: tabular-nums; min-width: 32px; text-align: center; ${isLowTime ? 'animation: pulse 1s infinite;' : ''}">${state.timeRemaining}s</span>
            `;
            if (isLowTime && !document.getElementById('eqe-pulse-animation')) {
                const style = document.createElement('style');
                style.id = 'eqe-pulse-animation';
                style.textContent = `@keyframes pulse { 0%, 100% { transform: scale(1); } 50% { transform: scale(1.1); } }`;
                document.head.appendChild(style);
            }
        } else if (state.currentPhase) {
            island.style.background = state.currentPhase === 'question' ? questionGradient : answerGradient;
            island.innerHTML = `
                <span style="margin-right: 8px; font-size: 18px;">${emoji}</span>
                <span style="opacity: 0.9; margin-right: 8px;">${phase}</span>
                <span style="font-size: 20px; font-weight: 700;">✓</span>
            `;
        }
    }

    function startCountdown(durationInSeconds) {
        stopCountdown();
        state.timeRemaining = durationInSeconds;
        updateDynamicIslandTimer();

        const intervalId = setInterval(() => {
            state.timeRemaining--;
            updateDynamicIslandTimer();
            if (state.timeRemaining < 0) { state.timeRemaining = 0; updateDynamicIslandTimer(); }
        }, 1000);

        state.countdownInterval = registerInterval(intervalId);
    }

    function stopCountdown() {
        if (state.countdownInterval) {
            clearInterval(state.countdownInterval);
            cleanupRegistry.intervals.delete(state.countdownInterval);
            state.countdownInterval = null;
        }
    }

    function hideTimer() {
        stopCountdown();
        state.timeRemaining = 0;
        state.currentPhase = null;
        updateDynamicIslandTimer();
    }

        panel.querySelector('#eqe-close-settings').addEventListener('click', toggleSettings);

        return panel;
    }

    function toggleSettings() {
        const panel = document.getElementById('eqe-auto-advance-settings');
        if (!panel) return;

        if (panel.style.display === 'none') {
            panel.style.display = 'block';

            // ── v8.3 FIX #3 ──────────────────────────────────────────────────
            // Attach a one-shot mousedown listener so clicking anywhere outside
            // the panel dismisses it, matching all other overlay behaviour.
            // We use mousedown (not click) to fire before focus shifts, and
            // setTimeout ensures the listener is added AFTER this event cycle
            // so the same click that opened the panel doesn't also close it.
            // ─────────────────────────────────────────────────────────────────
            const outsideClickHandler = (e) => {
                if (!panel.contains(e.target)) {
                    panel.style.display = 'none';
                    document.removeEventListener('mousedown', outsideClickHandler, true);
                }
            };
            setTimeout(() => {
                document.addEventListener('mousedown', outsideClickHandler, true);
            }, 0);

        } else {
            panel.style.display = 'none';
        }
    }

    function saveSettings() {
        const questionTimer = parseInt(document.getElementById('eqe-question-timer').value);
        const answerTimer = parseInt(document.getElementById('eqe-answer-timer').value);
        const autoAdvanceEnabled = document.getElementById('eqe-enable-autoadvance').checked;
        const autoSelectOnTimeout = document.getElementById('eqe-auto-select').checked;

        if (isNaN(questionTimer) || questionTimer < 1 || questionTimer > 300) {
            showToast('⚠️ Question timer must be between 1-300 seconds', 'warning');
            return;
        }
        if (isNaN(answerTimer) || answerTimer < 1 || answerTimer > 300) {
            showToast('⚠️ Answer timer must be between 1-300 seconds', 'warning');
            return;
        }

        const wasEnabled = config.autoAdvanceEnabled;
        config.questionTimer = questionTimer;
        config.answerTimer = answerTimer;
        config.autoAdvanceEnabled = autoAdvanceEnabled;
        config.autoSelectOnTimeout = autoSelectOnTimeout;

        GM_setValue('questionTimer', questionTimer);
        GM_setValue('answerTimer', answerTimer);
        GM_setValue('autoAdvanceEnabled', autoAdvanceEnabled);
        GM_setValue('autoSelectOnTimeout', autoSelectOnTimeout);

        showToast('✓ Settings saved successfully!', 'success');
        toggleSettings();

        const btnToggle = document.getElementById('eqe-btn-toggle');
        if (btnToggle) {
            if (autoAdvanceEnabled) {
                btnToggle.innerHTML = `<svg xmlns="http://www.w3.org/2000/svg" width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><rect x="6" y="4" width="4" height="16"/><rect x="14" y="4" width="4" height="16"/></svg>`;
                btnToggle.title = "Pause Auto-Advance";
            } else {
                btnToggle.innerHTML = `<svg xmlns="http://www.w3.org/2000/svg" width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><polygon points="5 3 19 12 5 21 5 3"/></svg>`;
                btnToggle.title = "Start Auto-Advance";
            }
        }

        if (autoAdvanceEnabled && !wasEnabled) {
            if (state.currentPhase) { startTimer(); } else { onQuestionLoad(); }
        } else if (!autoAdvanceEnabled && wasEnabled) {
            clearTimer();
            hideTimer();
        } else if (autoAdvanceEnabled && state.currentPhase) {
            startTimer();
        }
    }

    function showToast(message, type = 'info') {
        const existingToast = document.getElementById('eqe-toast');
        if (existingToast) existingToast.remove();

        const isDark = isDarkMode();
        const toast = document.createElement('div');
        toast.id = 'eqe-toast';

        const colors = {
            success: isDark ? '#10b981' : '#059669',
            warning: isDark ? '#f59e0b' : '#d97706',
            info: isDark ? '#3b82f6' : '#2563eb'
        };

        Object.assign(toast.style, {
            position: 'fixed', bottom: '20px', right: '20px',
            backgroundColor: isDark ? 'rgba(31, 41, 55, 0.95)' : 'rgba(255, 255, 255, 0.95)',
            color: isDark ? '#f3f4f6' : '#1f2937',
            padding: '12px 20px', borderRadius: '8px',
            border: `2px solid ${colors[type]}`,
            boxShadow: '0 4px 6px rgba(0, 0, 0, 0.1)',
            zIndex: '9999999', fontSize: '14px', fontWeight: '600',
            fontFamily: 'Arial, sans-serif',
            animation: 'slideIn 0.3s ease-out'
        });

        toast.textContent = message;
        document.body.appendChild(toast);

        if (!document.getElementById('eqe-toast-style')) {
            const style = document.createElement('style');
            style.id = 'eqe-toast-style';
            style.textContent = `@keyframes slideIn { from { transform: translateX(400px); opacity: 0; } to { transform: translateX(0); opacity: 1; } }`;
            document.head.appendChild(style);
        }

        const toastTimeoutId = registerTimeout(setTimeout(() => {
            toast.style.animation = 'slideIn 0.3s ease-out reverse';
            const removeTimeoutId = registerTimeout(setTimeout(() => {
                toast.remove();
                cleanupRegistry.timeouts.delete(removeTimeoutId);
            }, 300));
            cleanupRegistry.timeouts.delete(toastTimeoutId);
        }, 3000));
    }

    // ============================================================================
    // TIMER LOGIC
    // ============================================================================

    function clearTimer() {
        if (state.timerHandle) {
            clearTimeout(state.timerHandle);
            cleanupRegistry.timeouts.delete(state.timerHandle);
            state.timerHandle = null;
        }
    }

    function startTimer() {
        if (!config.autoAdvanceEnabled) { hideTimer(); return; }
        clearTimer();
        if (state.currentPhase === 'question') { startQuestionTimer(); }
        else if (state.currentPhase === 'answer') { startAnswerTimer(); }
    }

    function startQuestionTimer() {
        startCountdown(config.questionTimer);

        const timerHandle = setTimeout(() => {
            state.timeRemaining = 0;
            updateDynamicIslandTimer();

            if (config.autoSelectOnTimeout && !state.isAnswerSelected) {
                selectRandomAnswer();
                playNotificationSound();
                const checkTimeoutId = registerTimeout(setTimeout(() => {
                    const btn = getCheckBtn();
                    if (btn) {
                        btn.click();
                        clearTimer();
                        const phaseTimeoutId = registerTimeout(setTimeout(() => {
                            state.currentPhase = 'answer';
                            if (config.autoAdvanceEnabled) { startAnswerTimer(); } else { hideTimer(); }
                            cleanupRegistry.timeouts.delete(phaseTimeoutId);
                        }, 300));
                    }
                    cleanupRegistry.timeouts.delete(checkTimeoutId);
                }, 500));
            } else if (state.isAnswerSelected) {
                playNotificationSound();
                const checkTimeoutId = registerTimeout(setTimeout(() => {
                    const btn = getCheckBtn();
                    if (btn) {
                        btn.click();
                        clearTimer();
                        const phaseTimeoutId = registerTimeout(setTimeout(() => {
                            state.currentPhase = 'answer';
                            if (config.autoAdvanceEnabled) { startAnswerTimer(); } else { hideTimer(); }
                            cleanupRegistry.timeouts.delete(phaseTimeoutId);
                        }, 300));
                    }
                    cleanupRegistry.timeouts.delete(checkTimeoutId);
                }, 500));
            }

            cleanupRegistry.timeouts.delete(timerHandle);
        }, config.questionTimer * 1000);

        state.timerHandle = registerTimeout(timerHandle);
    }

    function startAnswerTimer() {
        startCountdown(config.answerTimer);

        const timerHandle = setTimeout(() => {
            state.timeRemaining = 0;
            updateDynamicIslandTimer();
            playNotificationSound();

            const hideTimeoutId = registerTimeout(setTimeout(() => {
                hideTimer();
                const nextBtn = getNextBtn();
                if (nextBtn) {
                    document.dispatchEvent(new KeyboardEvent('keydown', { key: 'ArrowRight', code: 'ArrowRight', bubbles: true }));
                }
                cleanupRegistry.timeouts.delete(hideTimeoutId);
            }, 800));

            cleanupRegistry.timeouts.delete(timerHandle);
        }, config.answerTimer * 1000);

        state.timerHandle = registerTimeout(timerHandle);
    }

    function selectRandomAnswer() {
        const answerButtons = getAnswerButtons();
        if (answerButtons.length > 0) {
            answerButtons[Math.floor(Math.random() * answerButtons.length)].click();
            state.isAnswerSelected = true;
            return;
        }
        const options = ['1', '2', '3', '4', '5'];
        document.dispatchEvent(new KeyboardEvent('keydown', {
            key: options[Math.floor(Math.random() * options.length)],
            bubbles: true
        }));
        state.isAnswerSelected = true;
    }

    // ============================================================================
    // QUESTION DETECTION
    // ============================================================================

    function onQuestionLoad() {
        if (state.questionLoadDebounce) {
            clearTimeout(state.questionLoadDebounce);
            cleanupRegistry.timeouts.delete(state.questionLoadDebounce);
        }

        const debounceId = setTimeout(() => {
            if (state.isProcessingNewQuestion) { return; }
            state.isProcessingNewQuestion = true;

            try {
                const currentQuestion = getQuestionText();
                const shouldForceTimerStart = state.manuallyNavigated;
                if (state.manuallyNavigated) state.manuallyNavigated = false;

                if (currentQuestion && (currentQuestion !== state.lastQuestionText || shouldForceTimerStart)) {
                    console.log('[Auto-Advance] New question detected');
                    state.lastQuestionText = currentQuestion;
                    state.currentPhase = 'question';
                    state.isAnswerSelected = false;
                    state.spacePressed = false;
                    state.answerSelectedAfterSpace = false;
                    if (config.autoAdvanceEnabled) { startTimer(); } else { hideTimer(); }
                }
            } catch (error) {
                console.error('[Auto-Advance] Error in onQuestionLoad:', error);
            } finally {
                const releaseTimeoutId = registerTimeout(setTimeout(() => {
                    state.isProcessingNewQuestion = false;
                    cleanupRegistry.timeouts.delete(releaseTimeoutId);
                }, 100));
            }

            cleanupRegistry.timeouts.delete(debounceId);
        }, 150);

        state.questionLoadDebounce = registerTimeout(debounceId);
    }

    // ============================================================================
    // KEYBOARD HANDLER
    // ============================================================================

    function handleKeydown(e) {
        if (['INPUT', 'TEXTAREA', 'SELECT'].includes(document.activeElement?.tagName)) {
            if (!document.activeElement.id.startsWith('eqe-')) return;
        }

        const key = e.key.toLowerCase();

        // Direct module selection on Dashboard
        if (/^[1-9]$/.test(key) && !e.shiftKey && !e.ctrlKey && !e.altKey) {
            if (window.location.href.includes('/dashboard')) {
                const courses = GM_getValue('saved_courses', []);
                const num = parseInt(key);
                if (num > 0 && num <= courses.length) {
                    window.location.href = courses[num - 1].url;
                    e.preventDefault();
                    e.stopImmediatePropagation();
                    return;
                }
            }
        }

        // ── v8.3 FIX #2 ─────────────────────────────────────────────────────
        // Pressing 0 no longer immediately navigates to the dashboard.
        // First press shows a confirmation toast; second press within 2s confirms.
        // The state resets automatically after 2s if not confirmed.
        // Shift+D retains the old instant-navigate behaviour (deliberate keycombo).
        // ────────────────────────────────────────────────────────────────────
        if (key === '0') {
            if (state.dashboardConfirmPending) {
                // Second press — confirmed, navigate now
                clearTimeout(state.dashboardConfirmTimeout);
                state.dashboardConfirmPending = false;
                state.dashboardConfirmTimeout = null;
                window.location.href = 'https://www.e-qe.online/dashboard';
            } else {
                // First press — ask for confirmation
                state.dashboardConfirmPending = true;
                showToast('🏠 Press 0 again to go to Dashboard', 'warning');
                state.dashboardConfirmTimeout = setTimeout(() => {
                    state.dashboardConfirmPending = false;
                    state.dashboardConfirmTimeout = null;
                }, 2000);
            }
            e.preventDefault();
            e.stopImmediatePropagation();
            return;
        }

        if (key === 'd' && e.shiftKey) {
            // Shift+D: instant navigate (deliberate combo, no confirmation needed)
            window.location.href = 'https://www.e-qe.online/dashboard';
            e.preventDefault();
            e.stopImmediatePropagation();
            return;
        }

        if (key === '6') {
            toggleSettings();
            e.preventDefault();
            e.stopImmediatePropagation();
            return;
        }

        if (key === '7') {
            const btnToggle = document.getElementById('eqe-btn-toggle');
            if (btnToggle) { toggleAutoAdvance(btnToggle); e.preventDefault(); e.stopImmediatePropagation(); }
            return;
        }

        if (key === '8') {
            cyclePreset();
            e.preventDefault();
            e.stopImmediatePropagation();
            return;
        }

        if (key === 'm' || key === '9') {
            showCourseSwitcher();
            e.preventDefault();
            e.stopImmediatePropagation();
            return;
        }

        if (e.key === 'Escape') {
            const settings = document.getElementById('eqe-auto-advance-settings');
            if (settings && settings.style.display !== 'none') {
                toggleSettings();
                e.preventDefault();
                e.stopImmediatePropagation();
            }
            return;
        }

        if (key === 't') {
            if (e.shiftKey) {
                showPresetTable();
                e.preventDefault();
                e.stopImmediatePropagation();
                return;
            }
            if (!state.tKeyPressTimer) {
                state.tKeyPressTimer = registerTimeout(setTimeout(() => {
                    showPresetTable();
                    state.tKeyLongPressed = true;
                }, 1000));
            }
            return;
        }

        if (key === 's' && e.shiftKey) {
            toggleSettings();
            e.preventDefault();
            e.stopImmediatePropagation();
            return;
        }

        if (key === 'c') {
            const cBtn = getCGroupBtn();
            if (cBtn) { cBtn.click(); e.preventDefault(); e.stopImmediatePropagation(); }
            return;
        }

        if (key === 'a' && e.shiftKey) {
            const btnToggle = document.getElementById('eqe-btn-toggle');
            if (btnToggle) { toggleAutoAdvance(btnToggle); e.preventDefault(); e.stopImmediatePropagation(); }
            return;
        }

        if (key === 'a' && !e.shiftKey) {
            const aBtn = getExplainBtn();
            if (aBtn) { aBtn.click(); e.preventDefault(); e.stopImmediatePropagation(); }
            return;
        }

        if (key === 'i') {
            const imageBtn = getViewImageBtn();
            if (imageBtn) { imageBtn.click(); e.preventDefault(); e.stopImmediatePropagation(); }
            return;
        }

        if (key === 'h') {
            toggleSidebar();
            e.preventDefault();
            e.stopImmediatePropagation();
            return;
        }

        if (key === '?' || (e.key === '?' && e.shiftKey)) {
            showShortcutsHelp();
            e.preventDefault();
            e.stopImmediatePropagation();
            return;
        }

        if (['1', '2', '3', '4', '5'].includes(key)) {
            state.isAnswerSelected = true;
            if (state.spacePressed && !state.answerSelectedAfterSpace) {
                state.answerSelectedAfterSpace = true;
            }
        }

        if (e.key === ' ' || e.key === 'Enter') {
            const btn = getCheckBtn();
            if (btn) {
                e.preventDefault();
                e.stopImmediatePropagation();

                if (!state.spacePressed) {
                    state.spacePressed = true;
                    if (!state.isAnswerSelected) { return; }
                    else {
                        btn.click();
                        clearTimer();
                        const phaseTimeoutId = registerTimeout(setTimeout(() => {
                            state.currentPhase = 'answer';
                            if (config.autoAdvanceEnabled) { startAnswerTimer(); } else { hideTimer(); }
                            cleanupRegistry.timeouts.delete(phaseTimeoutId);
                        }, 300));
                    }
                } else if (state.answerSelectedAfterSpace) {
                    btn.click();
                    clearTimer();
                    const phaseTimeoutId = registerTimeout(setTimeout(() => {
                        state.currentPhase = 'answer';
                        state.spacePressed = false;
                        state.answerSelectedAfterSpace = false;
                        if (config.autoAdvanceEnabled) { startAnswerTimer(); } else { hideTimer(); }
                        cleanupRegistry.timeouts.delete(phaseTimeoutId);
                    }, 300));
                } else {
                    btn.click();
                }
            }
            return;
        }

        if (!['ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown'].includes(e.key)) return;

        e.preventDefault();
        e.stopImmediatePropagation();

        clearTimer();
        stopCountdown();
        state.manuallyNavigated = true;

        if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
            getNextBtn()?.click();
            const navTimeoutId = registerTimeout(setTimeout(() => {
                state.lastQuestionText = null;
                onQuestionLoad();
                cleanupRegistry.timeouts.delete(navTimeoutId);
            }, 200));
        }
        if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
            getPrevBtn()?.click();
            const navTimeoutId = registerTimeout(setTimeout(() => {
                state.lastQuestionText = null;
                onQuestionLoad();
                cleanupRegistry.timeouts.delete(navTimeoutId);
            }, 200));
        }
    }

    function handleKeyup(e) {
        if (e.key.toLowerCase() === 't') {
            if (state.tKeyPressTimer) {
                clearTimeout(state.tKeyPressTimer);
                cleanupRegistry.timeouts.delete(state.tKeyPressTimer);
                state.tKeyPressTimer = null;
            }
            if (!state.tKeyLongPressed && !e.shiftKey) cyclePreset();
            state.tKeyLongPressed = false;
        }
    }

    // ============================================================================
    // INITIALIZATION
    // ============================================================================

    function init() {
        console.log('[Auto-Advance] Initializing v8.4...');

        if (!document.getElementById('eqe-auto-advance-settings')) {
            const settings = createSettingsPanel();
            document.body.appendChild(settings);
        }

        if (!document.getElementById('eqe-dynamic-island')) {
            const island = createDynamicIslandTimer();
            document.body.appendChild(island);
        }

        injectInlineControls();

        if (config.sidebarHidden) {
            const sidebar = document.querySelector('aside.h-full.lg\\:w-\\[270px\\], aside.h-full.xl\\:w-\\[300px\\]');
            if (sidebar) sidebar.style.display = 'none';
        }

        registerEventListener(document, 'keydown', handleKeydown, true);
        registerEventListener(document, 'keyup', handleKeyup, true);

        if (window.location.href.includes('/dashboard')) {
            scanAndSaveCourses();
            decorateDashboardModules();
        }

        const initTimeoutId = registerTimeout(setTimeout(() => {
            state.lastQuestionText = getQuestionText();
            if (config.autoAdvanceEnabled) onQuestionLoad();

            if (window.location.href.includes('/lesson/') || window.location.href.includes('/exam/')) {
                const courses = GM_getValue('saved_courses', []);
                const currentId = window.location.href.split('/').pop();
                const currentCourse = courses.find(c => c.id === currentId);
                if (currentCourse) {
                    showHUDToast({ emoji: '📍', name: 'Current Course', q: '-', a: '-', desc: currentCourse.name });
                }
            }

            cleanupRegistry.timeouts.delete(initTimeoutId);
        }, 500));
    }

    const observer = new MutationObserver(() => {
        if (!document.getElementById('eqe-inline-container')) injectInlineControls();
        if (window.location.href !== state.lastUrl) {
            state.lastUrl = window.location.href;
            injectInlineControls();
        }
        if (window.location.href.includes('/dashboard')) {
            scanAndSaveCourses();
            decorateDashboardModules();
        }
    });

    registerObserver(observer);
    registerEventListener(window, 'beforeunload', cleanup, false);
    registerEventListener(window, 'pagehide', cleanup, false);

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', () => {
            init();
            observer.observe(document.body, { childList: true, subtree: true });
        });
    } else {
        init();
        observer.observe(document.body, { childList: true, subtree: true });
    }

    // NOTE: fallback init — removed the immediate .delete() that was cancelling
    // cleanup registration in v8.2 (bug fix).
    registerTimeout(setTimeout(init, 1000));
})();
