    <audio id="paniyoloAudio" loop preload="auto" style="display:none;">
        <source src="https://raw.githubusercontent.com/achma-learning/eqe/main/Paniyolo%20-%20Coloring-LFpHsbRrK4M.mp3" type="audio/mpeg">
        Votre navigateur ne supporte pas l'audio HTML5.
    </audio>

    <script>
        const audio = document.getElementById('paniyoloAudio');
        const playBtn = document.getElementById('playBtn');
        const status = document.getElementById('status');
        let isPlaying = false;

        function togglePlay() {
            if (isPlaying) {
                audio.pause();
                playBtn.textContent = '▶️ PLAY';
                playBtn.style.background = '#ff6b6b';
                status.textContent = '⏸️ Musique en pause';
                isPlaying = false;
            } else {
                audio.play();
                playBtn.textContent = '⏸️ PAUSE';
                playBtn.style.background = '#4ecdc4';
                status.textContent = '🔊 Paniyolo Coloring en lecture...';
                isPlaying = true;
            }
        }

        // Gestion automatique des états
        audio.addEventListener('play', () => {
            playBtn.textContent = '⏸️ PAUSE';
            playBtn.style.background = '#4ecdc4';
            status.textContent = '🔊 Paniyolo Coloring en lecture...';
            isPlaying = true;
        });

        audio.addEventListener('pause', () => {
            playBtn.textContent = '▶️ PLAY';
            playBtn.style.background = '#ff6b6b';
            status.textContent = '⏸️ Musique en pause';
            isPlaying = false;
        });

        audio.addEventListener('ended', () => {
            // Redémarre automatiquement grâce à l'attribut loop
            status.textContent = '🔄 Boucle terminée - Redémarrage...';
        });
    </script>
