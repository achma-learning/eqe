// ==UserScript==
// @name         e-qe.online Diagnostic Script
// @namespace    https://e-qe.online/
// @version      0.1
// @description  Generates console output for diagnosing e-qe.online page structure and changes.
// @match        https://e-qe.online/*
// @match        https://www.e-qe.online/*
// @grant        none
// @run-at       document-idle
// ==/UserScript==

(() => {
    'use strict';

    console.log('[Diagnostic] Initializing diagnostic userscript...');

    const log = (message, ...args) => console.log(`[Diagnostic] ${new Date().toLocaleTimeString()} - ${message}`, ...args);
    const error = (message, ...args) => console.error(`[Diagnostic ERROR] ${new Date().toLocaleTimeString()} - ${message}`, ...args);

    // ============================================================================
    // CONFIGURATION
    // ============================================================================
    const POLLING_INTERVAL = 3000; // Check for elements every 3 seconds
    const OBSERVER_THROTTLE_MS = 500; // Throttle MutationObserver logs

    // ============================================================================
    // DOM SELECTORS (Reused from Auto-Advance script)
    // ============================================================================
    const getNextBtn = () => document.querySelector('button[aria-label*="next" i], button[aria-label="Next question" i]');
    const getPrevBtn = () => document.querySelector('button[aria-label*="previous" i], button[aria-label="Previous question" i]');
    const getCheckBtn = () => {
        const buttons = [...document.querySelectorAll('button')];
        return buttons.find(b => {
            const text = b.textContent?.trim();
            return /check\s*answer|submit|verify|show\s*answer/i.test(text) && !b.disabled;
        });
    };
    const getAnswerButtons = () => {
        const potentialAnswerContainers = document.querySelectorAll('div.select-none.border.p-3.shadow-md.cursor-pointer');
        
        return Array.from(potentialAnswerContainers).filter(container => {
            const firstChild = container.firstElementChild;
            if (firstChild && firstChild.textContent) {
                const text = firstChild.textContent.trim();
                return ['A', 'B', 'C', 'D', 'E'].includes(text);
            }
            return false;
        });
    };
    const getQuestionTextElement = () => document.querySelector('h1, h2, [class*="question"]');

    // ============================================================================
    // STATE
    // ============================================================================
    let lastKnownURL = window.location.href;
    let lastQuestionText = null;
    let observerTimeout = null;

    // ============================================================================
    // FUNCTIONS
    // ============================================================================

    function checkURLChange() {
        if (window.location.href !== lastKnownURL) {
            log(`URL changed from: ${lastKnownURL} to: ${window.location.href}`);
            lastKnownURL = window.location.href;
            // Force a re-check of elements after URL change
            setTimeout(checkElements, 100); 
        }
    }

    function checkElements() {
        log('--- Element Status Check ---');

        const questionElement = getQuestionTextElement();
        const currentQuestion = questionElement ? questionElement.textContent.trim() : null;

        if (currentQuestion && currentQuestion !== lastQuestionText) {
            log(`NEW QUESTION DETECTED: "${currentQuestion}"`);
            lastQuestionText = currentQuestion;
        } else if (currentQuestion) {
            // log(`Question: "${currentQuestion}" (no change)`); // Too noisy, only log new
        } else {
            log('Question element NOT found.');
        }

        const answerButtons = getAnswerButtons();
        if (answerButtons.length > 0) {
            log(`Found ${answerButtons.length} answer button(s). First: "${answerButtons[0].textContent?.trim()}"`);
            // log details of each button for more info
            answerButtons.forEach((btn, idx) => {
                log(`  Answer Button ${idx + 1}: Text="${btn.textContent?.trim()}", Class="${btn.className}", Role="${btn.getAttribute('role')}"`);
            });
        } else {
            log('Answer buttons NOT found.');
        }

        const checkBtn = getCheckBtn();
        if (checkBtn) {
            log(`Check/Submit button found: Text="${checkBtn.textContent?.trim()}", Disabled=${checkBtn.disabled}`);
        } else {
            log('Check/Submit button NOT found.');
        }

        const nextBtn = getNextBtn();
        if (nextBtn) {
            log(`Next button found: Text="${nextBtn.textContent?.trim()}", Disabled=${nextBtn.disabled}`);
        } else {
            log('Next button NOT found.');
        }

        const prevBtn = getPrevBtn();
        if (prevBtn) {
            log(`Previous button found: Text="${prevBtn.textContent?.trim()}", Disabled=${prevBtn.disabled}`);
        } else {
            log('Previous button NOT found.');
        }
        log('--- End Element Status Check ---');
    }

    function setupMutationObserver() {
        const observer = new MutationObserver((mutations) => {
            // Throttle logging to prevent overwhelming the console
            if (observerTimeout) clearTimeout(observerTimeout);
            observerTimeout = setTimeout(() => {
                log('DOM changes detected (throttled). Checking elements...');
                checkElements();
                checkURLChange();
            }, OBSERVER_THROTTLE_MS);
        });

        // Observe the entire document body for changes in its children and subtree
        observer.observe(document.body, { childList: true, subtree: true, attributes: true });
        log('MutationObserver started for document.body.');
    }

    // ============================================================================
    // INITIALIZATION
    // ============================================================================
    function initDiagnostic() {
        log('Diagnostic userscript loaded. Setting up observers and polls.');

        // Initial check
        checkElements();
        checkURLChange();

        // Setup periodic polling for elements (in case MutationObserver misses something or for active checking)
        setInterval(checkElements, POLLING_INTERVAL);
        setInterval(checkURLChange, 1000); // Check URL more frequently

        // Setup MutationObserver for dynamic content changes
        setupMutationObserver();
    }

    // Wait for the DOM to be fully loaded
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initDiagnostic);
    } else {
        initDiagnostic();
    }

    // Fallback for very dynamic pages or if DOMContentLoaded is too late
    setTimeout(initDiagnostic, 1500);
})();
