    function playNotificationSound() {
        try {
            const audioContext = new (window.AudioContext || window.webkitAudioContext)();
            const oscillator = audioContext.createOscillator();
            const gainNode = audioContext.createGain();

            oscillator.connect(gainNode);
            gainNode.connect(audioContext.destination);

            oscillator.type = 'sine';
            oscillator.frequency.setValueAtTime(440, audioContext.currentTime);
            gainNode.gain.setValueAtTime(0, audioContext.currentTime);
            gainNode.gain.linearRampToValueAtTime(0.3, audioContext.currentTime + 0.01);
            oscillator.start(audioContext.currentTime);
            gainNode.gain.exponentialRampToValueAtTime(0.001, audioContext.currentTime + 0.5);
            oscillator.stop(audioContext.currentTime + 0.5);
        } catch (e) {
            console.warn('[Auto-Advance] Could not play notification sound:', e);
        }
    }

    function updateDynamicIslandTimer() {
        const island = document.getElementById('eqe-dynamic-island');
        if (!island) return;

        // Gradient colors for question and answer phases (re-defined here for local scope)
        const questionGradient = island.dataset.questionGradient;
        const answerGradient = island.dataset.answerGradient;
        const lowTimeGradient = 'linear-gradient(135deg, #ef4444 0%, #dc2626 100%)'; // Tailwind red-500 / red-700
        const neutralGradient = 'linear-gradient(135deg, #4b5563 0%, #374151 100%)'; // Neutral gray for when phase not set (if needed)

        // Only show when auto-advance is enabled and we have an active phase
        if (!config.autoAdvanceEnabled || !state.currentPhase) {
            // Fade out animation
            island.style.opacity = '0';
            island.style.transform = 'translateX(-50%) scale(0.8)';
            setTimeout(() => {
                island.style.display = 'none';
                // Remove the pulse animation if it exists and timer is hidden
                const pulseStyle = document.getElementById('eqe-pulse-animation');
                if (pulseStyle) pulseStyle.remove();
            }, 400); // Allow transition to complete
            return;
        }

        // Show island with fade in
        if (island.style.display === 'none') {
            island.style.display = 'flex';
            setTimeout(() => {
                island.style.opacity = '1';
                island.style.transform = 'translateX(-50%) scale(1)';
            }, 10); // Small delay to allow display:flex to apply before transition
        }

        // Determine current phase and base gradient
        const phase = state.currentPhase === 'question' ? 'Question' : 'Answer';
        let currentBackgroundGradient = state.currentPhase === 'question' ? questionGradient : answerGradient;
        
        // Emoji indicators
        const emoji = state.currentPhase === 'question' ? '❓' : '✅';
        
        // Create pulsing effect and red styling when time is running low
        const isLowTime = state.timeRemaining <= 3 && state.timeRemaining > 0;
        if (isLowTime) {
            currentBackgroundGradient = lowTimeGradient; // Override to red gradient for low time
        }
        
        island.style.background = currentBackgroundGradient;
        
        if (state.timeRemaining > 0) {
            const timeSpanColor = isLowTime ? 'red' : 'white'; // Seconds color to red if low time

            island.innerHTML = `
                <span style="margin-right: 8px; font-size: 18px;">${emoji}</span>
                <span style="opacity: 0.9; margin-right: 8px;">${phase}</span>
                <span style="
                    font-size: 20px;
                    font-weight: 700;
                    font-variant-numeric: tabular-nums;
                    min-width: 32px;
                    text-align: center;
                    font-family: 'Cambria Math', monospace;
                    color: ${timeSpanColor};
                    ${isLowTime ? 'animation: pulse 1s infinite;' : ''}
                ">${state.timeRemaining}s</span>
            `;
            
            // Add pulsing animation for low time if not already added
            if (isLowTime && !document.getElementById('eqe-pulse-animation')) {
                const style = document.createElement('style');
                style.id = 'eqe-pulse-animation';
                style.textContent = `
                    @keyframes pulse {
                        0% { transform: scale(1); opacity: 1; }
                        50% { transform: scale(1.1); opacity: 0.7; }
                        100% { transform: scale(1); opacity: 1; }
                    }
                `;
                document.head.appendChild(style);
            } else if (!isLowTime && document.getElementById('eqe-pulse-animation')) {
                // Remove pulse animation if time is no longer low
                document.getElementById('eqe-pulse-animation').remove();
            }

        } else if (state.currentPhase) {
            // Show completion state briefly for 0s
            // Background should revert to phase-specific, not lowTime red
            island.style.background = state.currentPhase === 'question' ? questionGradient : answerGradient;
            if (document.getElementById('eqe-pulse-animation')) {
                document.getElementById('eqe-pulse-animation').remove();
            }

            island.innerHTML = `
                <span style="margin-right: 8px; font-size: 18px;">${emoji}</span>
                <span style="opacity: 0.9; margin-right: 8px;">${phase}</span>
                <span style="font-size: 20px; font-weight: 700; font-family: 'Cambria Math', monospace;">✓</span>
            `;
        }

        // Expand island width based on content - needs to be done *after* innerHTML is set
        // Use a temporary element to measure width accurately
        const tempSpan = document.createElement('span');
        tempSpan.style.cssText = `
            visibility: hidden;
            position: absolute;
            white-space: nowrap;
            font-size: 15px; /* Match island font size */
            font-weight: 600; /* Match island font weight */
            padding: 0 20px; /* Match island padding */
            font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif;
            /* Include dynamic font family for accurate measurement if it affects width */
            ${isLowTime ? `font-family: 'Cambria Math', monospace;` : ''} 
        `;
        tempSpan.innerHTML = `
            <span style="margin-right: 8px; font-size: 18px;">${emoji}</span>
            <span style="opacity: 0.9; margin-right: 8px;">${phase}</span>
            <span style="
                font-size: 20px;
                font-weight: 700;
                font-variant-numeric: tabular-nums;
                min-width: 32px;
                text-align: center;
                font-family: 'Cambria Math', monospace;
                color: ${isLowTime ? 'red' : 'white'};
            ">${state.timeRemaining > 0 ? state.timeRemaining + 's' : '✓'}</span>
        `;
        document.body.appendChild(tempSpan);
        
        const measuredWidth = tempSpan.offsetWidth;
        document.body.removeChild(tempSpan);
        
        island.style.minWidth = `${Math.max(180, measuredWidth + 40)}px`; // +40 for padding on both sides
    }

    function startCountdown(durationInSeconds) {
        stopCountdown();
        
        state.timeRemaining = durationInSeconds;
        updateDynamicIslandTimer();
        
        state.countdownInterval = setInterval(() => {
            state.timeRemaining--;
            updateDynamicIslandTimer();
            
            if (state.timeRemaining < 0) {
                state.timeRemaining = 0;
                stopCountdown(); // Stop countdown when it hits 0
            }
        }, 1000);
    }

    function stopCountdown() {
        if (state.countdownInterval) {
            clearInterval(state.countdownInterval);
            state.countdownInterval = null;
        }
        // Ensure pulse animation is removed if countdown stops
        const pulseStyle = document.getElementById('eqe-pulse-animation');
        if (pulseStyle) pulseStyle.remove();
    }

    function hideTimer() {
        stopCountdown();
        state.timeRemaining = 0;
        state.currentPhase = null;
        updateDynamicIslandTimer(); // This call will now hide the island
    }

    function makeDraggable(element) {
        let isDragging = false;
        let startX, startY, initialX, initialY;
        let clickTime = 0;

        element.addEventListener('mousedown', (e) => {
            const now = Date.now();
            if (now - clickTime < 300) return;
            clickTime = now;
            
            isDragging = true;
            startX = e.clientX;
            startY = e.clientY;
            
            const rect = element.getBoundingClientRect();
            initialX = rect.left;
            initialY = rect.top;
            
            element.style.cursor = 'grabbing';
            e.preventDefault();
        });

        document.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            
            const dx = e.clientX - startX;
            const dy = e.clientY - startY;
            
            element.style.left = (initialX + dx) + 'px';
            element.style.top = (initialY + dy) + 'px';
            element.style.right = 'auto';
        });

        document.addEventListener('mouseup', () => {
            if (isDragging) {
                isDragging = false;
                element.style.cursor = 'move';
            }
        });
    }

                        <span style="color: ${colors.labelColor};">Toggle Official/Community</span>
                    </div>
                    <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 6px;">
                        <span><kbd style="background: ${colors.kbdBg}; border: 1px solid ${colors.kbdBorder}; padding: 2px 6px; border-radius: 3px; font-family: monospace; color: ${colors.kbdText};">A</kbd></span>
                        <span style="color: ${colors.labelColor};">Open Explain</span>
                    </div>
                    <div style="display: flex; justify-content: space-between; align-items: center;">
                        <span><kbd style="background: ${colors.kbdBg}; border: 1px solid ${colors.kbdBorder}; padding: 2px 6px; border-radius: 3px; font-family: monospace; color: ${colors.kbdText};">Shift</kbd>+<kbd style="background: ${colors.kbdBg}; border: 1px solid ${colors.kbdBorder}; padding: 2px 6px; border-radius: 3px; font-family: monospace; color: ${colors.kbdText};">A</kbd></span>
                        <span style="color: ${colors.labelColor};">Pause/Resume Auto-Advance</span>
                    </div>
                </div>
            </div>
        `;

        panel.querySelector('#eqe-save-settings').addEventListener('click', saveSettings);
        panel.querySelector('#eqe-close-settings').addEventListener('click', toggleSettings);
        
        return panel;
    }

    function toggleSettings() {
        const panel = document.getElementById('eqe-auto-advance-settings');
        if (panel.style.display === 'none') {
            panel.style.display = 'block';
        } else {
            panel.style.display = 'none';
        }
    }

    function saveSettings() {
        const questionTimer = parseInt(document.getElementById('eqe-question-timer').value);
        const answerTimer = parseInt(document.getElementById('eqe-answer-timer').value);
        const autoAdvanceEnabled = document.getElementById('eqe-enable-autoadvance').checked;

        if (questionTimer >= 1 && answerTimer >= 1) {
            const wasEnabled = config.autoAdvanceEnabled;
            
            config.questionTimer = questionTimer;
            config.answerTimer = answerTimer;
            config.autoAdvanceEnabled = autoAdvanceEnabled;

            GM_setValue('questionTimer', questionTimer);
            GM_setValue('answerTimer', answerTimer);
            GM_setValue('autoAdvanceEnabled', autoAdvanceEnabled);

            showToast('✓ Settings saved successfully!', 'success');
            toggleSettings();
            
            if (autoAdvanceEnabled && !wasEnabled) {
                if (state.currentPhase) {
                    startTimer();
                } else {
                    onQuestionLoad();
                }
            } else if (!autoAdvanceEnabled && wasEnabled) {
                clearTimer();
                hideTimer();
            } else if (autoAdvanceEnabled && state.currentPhase) {
                startTimer();
            }
        } else {
            showToast('⚠️ Timers must be at least 1 second', 'warning');
        }
    }

    function showToast(message, type = 'info') {
        const existingToast = document.getElementById('eqe-toast');
        if (existingToast) {
            existingToast.remove();
        }

        const isDark = isDarkMode();
        const toast = document.createElement('div');
        toast.id = 'eqe-toast';

        const colors = {
            success: isDark ? '#10b981' : '#059669',
            warning: isDark ? '#f59e0b' : '#d97706',
            info: isDark ? '#3b82f6' : '#2563eb'
        };

        Object.assign(toast.style, {
            position: 'fixed',
            bottom: '20px',
            right: '20px',
            backgroundColor: isDark ? 'rgba(31, 41, 55, 0.95)' : 'rgba(255, 255, 255, 0.95)',
            color: isDark ? '#f3f4f6' : '#1f2937',
            padding: '12px 20px',
            borderRadius: '8px',
            border: `2px solid ${colors[type]}`,
            boxShadow: '0 4px 6px rgba(0, 0, 0, 0.1)',
            zIndex: '9999999',
            fontSize: '14px',
            fontWeight: '600',
            fontFamily: 'Arial, sans-serif',
            animation: 'slideIn 0.3s ease-out'
        });

        toast.textContent = message;
        document.body.appendChild(toast);

        if (!document.getElementById('eqe-toast-style')) {
            const style = document.createElement('style');
            style.id = 'eqe-toast-style';
            style.textContent = `
                @keyframes slideIn {
                    from {
                        transform: translateX(400px);
                        opacity: 0;
                    }
                    to {
                        transform: translateX(0);
                        opacity: 1;
                    }
                }
            `;
            document.head.appendChild(style);
        }

        setTimeout(() => {
            toast.style.animation = 'slideIn 0.3s ease-out reverse';
            setTimeout(() => toast.remove(), 300);
        }, 3000);
    }

    // ============================================================================ 
    // TIMER LOGIC
    // ============================================================================ 
    
    function clearTimer() {
        if (state.timerHandle) {
            clearTimeout(state.timerHandle);
            state.timerHandle = null;
        }
    }

    function startTimer() {
        if (!config.autoAdvanceEnabled) {
            hideTimer();
            return;
        }
        
        clearTimer();
        
        if (state.currentPhase === 'question') {
            startQuestionTimer();
        } else if (state.currentPhase === 'answer') {
            startAnswerTimer();
        }
    }

    function startQuestionTimer() {
        startCountdown(config.questionTimer);
        
        state.timerHandle = setTimeout(() => {
            state.timeRemaining = 0;
            updateDynamicIslandTimer();
            
            if (!state.isAnswerSelected) {
                selectRandomAnswer(); 
            }

            if (state.isAnswerSelected) {
                playNotificationSound();
                setTimeout(() => {
                    const btn = getCheckBtn();
                    if (btn) {
                        const spaceKeyEvent = new KeyboardEvent('keydown', {
                            key: ' ',
                            code: 'Space',
                            bubbles: true
                        });
                        document.dispatchEvent(spaceKeyEvent);
                    }
                }, 500);
            }
        }, config.questionTimer * 1000);
    }

    function startAnswerTimer() {
        startCountdown(config.answerTimer);
        
        state.timerHandle = setTimeout(() => {
            state.timeRemaining = 0;
            updateDynamicIslandTimer();

            playNotificationSound();
            
            setTimeout(() => {
                hideTimer();
                
                const nextBtn = getNextBtn();
                if (nextBtn) {
                    const arrowRightEvent = new KeyboardEvent('keydown', {
                        key: 'ArrowRight',
                        code: 'ArrowRight',
                        bubbles: true
                    });
                    document.dispatchEvent(arrowRightEvent);
                }
            }, 800);
        }, config.answerTimer * 1000);
    }

    function selectRandomAnswer() {
        const answerButtons = getAnswerButtons();
        
        if (answerButtons.length > 0) {
            const randomIndex = Math.floor(Math.random() * answerButtons.length);
            const selectedButton = answerButtons[randomIndex];
            selectedButton.click();
            
            state.isAnswerSelected = true;
            return;
        }

        const options = ['1', '2', '3', '4', '5'];
        const randomOption = options[Math.floor(Math.random() * options.length)];
        
        const keyEvent = new KeyboardEvent('keydown', {
            key: randomOption,
            code: `Digit${randomOption}`,
            bubbles: true
        });
        document.dispatchEvent(keyEvent);
        
        state.isAnswerSelected = true;
    }

    function onQuestionLoad() {
        if (state.isProcessingNewQuestion) {
            return;
        }
        state.isProcessingNewQuestion = true;

        try {
            const currentQuestion = getQuestionText();
            
            const shouldForceTimerStart = state.manuallyNavigated; 
            if (state.manuallyNavigated) {
                state.manuallyNavigated = false; 
            }

            if (currentQuestion && (currentQuestion !== state.lastQuestionText || shouldForceTimerStart)) {
                state.lastQuestionText = currentQuestion;
                state.currentPhase = 'question';
                state.isAnswerSelected = false;
                state.spacePressed = false;
                state.answerSelectedAfterSpace = false;                
                
                if (config.autoAdvanceEnabled) {
                    startTimer();
                } else {
                    hideTimer();
                }
            }
        } finally {
            state.isProcessingNewQuestion = false;
        }
    }

    // ============================================================================ 
    // KEYBOARD HANDLER
    // ============================================================================ 
    
    function handleKeydown(e) {
        if (['INPUT', 'TEXTAREA', 'SELECT'].includes(document.activeElement?.tagName)) {
            if (!document.activeElement.id.startsWith('eqe-')) return;
        }

        const key = e.key.toLowerCase();

        // --- NEW SHORTCUT LOGIC ---
        // 'C' for Official/Community
        if (key === 'c') {
            const cBtn = getCGroupBtn();
            if (cBtn) {
                cBtn.click();
                e.preventDefault();
                e.stopImmediatePropagation();
            }
            return;
        }

        // Shift + S to open/close settings panel
        if (key === 's' && e.shiftKey) {
            toggleSettings();
            e.preventDefault();
            e.stopImmediatePropagation();
            return;
        }

        // Shift + P to Pause/Resume Auto-Advance
        if (key === 'p' && e.shiftKey) {
            const autoAdvanceCheckbox = document.getElementById('eqe-enable-autoadvance');
            if (autoAdvanceCheckbox) {
                const newState = !autoAdvanceCheckbox.checked;
                autoAdvanceCheckbox.checked = newState;
                
                config.autoAdvanceEnabled = newState;
                GM_setValue('autoAdvanceEnabled', newState);
                
                const statusText = newState ? 'resumed ▶️' : 'paused ⏸️';
                showToast(`Auto-Advance ${statusText}`, newState ? 'success' : 'info');
                
                if (newState) {
                    if (state.currentPhase) {
                        startTimer();
                    } else {
                        onQuestionLoad();
                    }
                } else {
                    clearTimer();
                    hideTimer();
                }
                
                e.preventDefault();
                e.stopImmediatePropagation();
            }
            return;
        }

        if (key === 'a' && !e.shiftKey) {
            const aBtn = getExplainBtn();
            if (aBtn) {
                aBtn.click();
                e.preventDefault();
                e.stopImmediatePropagation();
            }
            return;
        }

        if (['1', '2', '3', '4', '5'].includes(key)) {
            state.isAnswerSelected = true;
            
            if (state.spacePressed && !state.answerSelectedAfterSpace) {
                state.answerSelectedAfterSpace = true;
            }
        }

        if (e.key === ' ' || e.key === 'Enter') {
            const btn = getCheckBtn();
            if (btn) {
                e.preventDefault();
                e.stopImmediatePropagation();
                
                if (!state.spacePressed) {
                    state.spacePressed = true;
                    
                    if (!state.isAnswerSelected) {
                        return;
                    } else {
                        btn.click();
                        
                        clearTimer();
                        setTimeout(() => {
                            state.currentPhase = 'answer';
                            if (config.autoAdvanceEnabled) {
                                startAnswerTimer();
                            } else {
                                hideTimer();
                            }
                        }, 300);
                    }
                } else if (state.answerSelectedAfterSpace) {
                    btn.click();
                    
                    clearTimer();
                    setTimeout(() => {
                        state.currentPhase = 'answer';
                        state.spacePressed = false;
                        state.answerSelectedAfterSpace = false;
                        if (config.autoAdvanceEnabled) {
                            startAnswerTimer();
                        } else {
                            hideTimer();
                        }
                    }, 300);
                } else {
                    btn.click();
                }
            }
            return;
        }

        if (!['ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown'].includes(e.key)) return;

        e.preventDefault();
        e.stopImmediatePropagation();

        clearTimer();
        stopCountdown();

        state.manuallyNavigated = true;

        if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
            getNextBtn()?.click();
            setTimeout(() => {
                state.lastQuestionText = null;
                onQuestionLoad();
            }, 200);
        }
        if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
            getPrevBtn()?.click();
            setTimeout(() => {
                state.lastQuestionText = null;
                onQuestionLoad();
            }, 200);
        }
    }

    // ============================================================================
    // INITIALIZATION
    // ============================================================================
    
    function init() {
        if (document.getElementById('eqe-auto-advance-gear')) return;

        const gear = createGearIcon();
        const settings = createSettingsPanel();
        const island = createDynamicIslandTimer();
        
        document.body.appendChild(gear);
        document.body.appendChild(settings);
        document.body.appendChild(island);

        document.addEventListener('keydown', handleKeydown, true);

        setTimeout(() => {
            state.lastQuestionText = getQuestionText();
            if (config.autoAdvanceEnabled) {
                onQuestionLoad();
            }
        }, 500);
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
    } else {
        init();
    }

    setTimeout(init, 1000);
})();
